cmake_minimum_required(VERSION 3.22.1)
project(dsh_pairing LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(boringssl REQUIRED CONFIG)
add_library(dsh_pairing SHARED adb_pairing.cpp)
target_compile_definitions(dsh_pairing PRIVATE NO_LOG)
target_compile_options(dsh_pairing PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_options(dsh_pairing PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
target_link_libraries(dsh_pairing PRIVATE boringssl::crypto_static log)
